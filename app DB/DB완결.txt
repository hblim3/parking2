CREATE TABLE `apartments` (
  `a_no` int NOT NULL AUTO_INCREMENT,
  `a_name` varchar(30) NOT NULL,
  `a_pwd` varchar(30) NOT NULL,
  `a_address` varchar(255) NOT NULL COMMENT '아파트 기본 주소',
  `a_detail_address` varchar(255) DEFAULT NULL COMMENT '상세 주소',
  PRIMARY KEY (`a_no`),
  UNIQUE KEY `a_name` (`a_name`),
  UNIQUE KEY `a_pwd` (`a_pwd`)
) ENGINE=InnoDB AUTO_INCREMENT=4 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE `car` (
  `c_no` int NOT NULL AUTO_INCREMENT,
  `c_name` varchar(50) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '사용자 지정 차량 이름',
  `c_number` varchar(255) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '차량 번호',
  `c_kind` varchar(30) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '차종',
  `c_note` varchar(255) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '관리자 메모',
  `c_date` datetime(6) DEFAULT CURRENT_TIMESTAMP(6) COMMENT '등록 날짜',
  `u_no` int DEFAULT NULL COMMENT '소유주 회원 번호',
  PRIMARY KEY (`c_no`),
  KEY `fk_car_user` (`u_no`),
  CONSTRAINT `fk_car_user` FOREIGN KEY (`u_no`) REFERENCES `user` (`u_no`) ON DELETE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=3 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `device_info` (
  `device_id` varchar(100) NOT NULL,
  `u_no` int NOT NULL,
  `fcm_token` varchar(255) DEFAULT NULL,
  `os_type` varchar(20) DEFAULT NULL,
  `last_login` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`device_id`),
  KEY `FK_device_user` (`u_no`),
  CONSTRAINT `FK_device_user` FOREIGN KEY (`u_no`) REFERENCES `user` (`u_no`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE `notifications` (
  `noti_no` int NOT NULL AUTO_INCREMENT,
  `u_no` int NOT NULL COMMENT '수신자 회원 번호',
  `noti_type` varchar(20) COLLATE utf8mb4_unicode_ci DEFAULT 'system' COMMENT '알림 종류',
  `noti_title` varchar(100) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '알림 제목',
  `noti_message` varchar(255) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '알림 상세 내용',
  `is_read` tinyint(1) DEFAULT '0' COMMENT '읽음 여부',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`noti_no`),
  KEY `FK_noti_user` (`u_no`),
  CONSTRAINT `FK_noti_user` FOREIGN KEY (`u_no`) REFERENCES `user` (`u_no`) ON DELETE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=4 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `parking_lot` (
  `pl_no` int NOT NULL AUTO_INCREMENT,
  `a_no` int NOT NULL COMMENT '아파트 번호',
  `pl_name` varchar(255) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '주차장 이름 (예: 명학아파트 지하주차장)',
  `pl_floor` varchar(20) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '층수 (예: B1)',
  `total_spaces` int NOT NULL COMMENT '전체 주차면 수',
  `used_spaces` int NOT NULL COMMENT '사용 중인 주차면 수',
  PRIMARY KEY (`pl_no`),
  KEY `fk_parking_lot_apartment` (`a_no`),
  CONSTRAINT `fk_parking_lot_apartment` FOREIGN KEY (`a_no`) REFERENCES `apartments` (`a_no`) ON DELETE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=2 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `parking_zone` (
  `pz_no` int NOT NULL AUTO_INCREMENT,
  `pl_no` int NOT NULL COMMENT '소속된 주차장 구역 번호',
  `area_number` varchar(255) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '주차면 이름 (예: A-1, 통로 1)',
  `location` varchar(255) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '상세 위치 설명',
  `status` varchar(20) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '상태 (사용가능, 사용중, 불법주차 등)',
  `layout_row` int DEFAULT NULL COMMENT '앱 화면 도면 행 위치',
  `layout_column` int DEFAULT NULL COMMENT '앱 화면 도면 열 위치',
  `status_change_reason` varchar(255) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '상태 변경 사유',
  `current_car_number` varchar(255) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '현재 주차된 차량 번호',
  PRIMARY KEY (`pz_no`),
  KEY `fk_parking_zone_parking_lot` (`pl_no`),
  CONSTRAINT `fk_parking_zone_parking_lot` FOREIGN KEY (`pl_no`) REFERENCES `parking_lot` (`pl_no`) ON DELETE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=10 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `registered_cars` (
  `v_no` int NOT NULL AUTO_INCREMENT COMMENT '방문객 차량 등록 번호',
  `u_no` int NOT NULL COMMENT '초대한 입주민의 회원 번호',
  `c_number` varchar(50) NOT NULL COMMENT '방문객 차량 번호',
  `reg_time` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '어플에서 등록한 시간',
  `park_time` datetime DEFAULT NULL COMMENT '실제 주차장 입차 시간',
  `expire_date` datetime DEFAULT NULL COMMENT '주차 만료 일시 (입차 시 24시간 뒤로 계산)',
  PRIMARY KEY (`v_no`),
  KEY `fk_visitor_user` (`u_no`),
  CONSTRAINT `fk_visitor_user` FOREIGN KEY (`u_no`) REFERENCES `user` (`u_no`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE `resident_inquiry` (
  `inquiry_no` int NOT NULL AUTO_INCREMENT,
  `u_no` int NOT NULL COMMENT '작성자 회원 번호',
  `c_no` int DEFAULT NULL COMMENT '관련 차량 번호 (선택사항)',
  `title` varchar(100) NOT NULL COMMENT '제목',
  `content` text NOT NULL COMMENT '내용',
  `status` varchar(20) DEFAULT 'pending' COMMENT '상태 (pending, answered)',
  `answer` text COMMENT '관리자 답변 내용',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '작성일시',
  `answered_at` datetime DEFAULT NULL COMMENT '답변일시',
  PRIMARY KEY (`inquiry_no`),
  KEY `FK_resident_inquiry_user` (`u_no`),
  CONSTRAINT `FK_resident_inquiry_user` FOREIGN KEY (`u_no`) REFERENCES `user` (`u_no`) ON DELETE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=2 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE `settings` (
  `setting_no` int NOT NULL AUTO_INCREMENT,
  `device_id` varchar(100) NOT NULL,
  `alert_push` tinyint(1) DEFAULT '1',
  `theme_mode` varchar(20) DEFAULT 'light',
  PRIMARY KEY (`setting_no`),
  UNIQUE KEY `device_id` (`device_id`),
  CONSTRAINT `FK_settings_device` FOREIGN KEY (`device_id`) REFERENCES `device_info` (`device_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE `user` (
  `u_no` int NOT NULL AUTO_INCREMENT,
  `u_id` varchar(20) NOT NULL,
  `u_pwd` varchar(255) NOT NULL COMMENT '암호화된 비밀번호 저장용',
  `p_date` datetime DEFAULT CURRENT_TIMESTAMP,
  `u_date` datetime DEFAULT CURRENT_TIMESTAMP,
  `u_dong` varchar(20) NOT NULL,
  `u_ho` varchar(20) NOT NULL,
  `a_no` int NOT NULL,
  `u_name` varchar(30) NOT NULL COMMENT '사용자 실명',
  `u_email` varchar(255) NOT NULL COMMENT '이메일 주소',
  `u_phone` varchar(20) DEFAULT NULL COMMENT '전화번호',
  `approval_status` varchar(20) NOT NULL DEFAULT 'pending' COMMENT '승인 상태 (pending, approved, rejected)',
  `reject_reason` varchar(255) DEFAULT NULL COMMENT '가입 거절 사유',
  PRIMARY KEY (`u_no`),
  UNIQUE KEY `u_id` (`u_id`),
  KEY `FK_user_apartment` (`a_no`),
  CONSTRAINT `FK_user_apartment` FOREIGN KEY (`a_no`) REFERENCES `apartments` (`a_no`) ON DELETE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=9 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE `waiting_list` (
